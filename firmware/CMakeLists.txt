cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD pico)   # RP2040 (Pro Micro RP2040 dung chung cau hinh nay: boot2 W25Q080, flash >= 2MB)
include(pico_sdk_import.cmake)
project(macropad15 C CXX ASM)
pico_sdk_init()
add_executable(macropad15 main.c usb_descriptors.c)
target_include_directories(macropad15 PRIVATE ${CMAKE_CURRENT_LIST_DIR})
# Man hinh OLED: mac dinh 128x64 (0.96"). Dung 0.91" 128x32 thi build voi: cmake .. -DOLED_H=32
if(DEFINED OLED_H)
  target_compile_definitions(macropad15 PRIVATE OLED_H=${OLED_H})
endif()
target_link_libraries(macropad15 pico_stdlib pico_flash hardware_flash hardware_i2c tinyusb_device tinyusb_board)
# Khong dung UART/USB stdio: GP0, GP1 la chan hang cua ma tran
pico_enable_stdio_uart(macropad15 0)
pico_enable_stdio_usb(macropad15 0)
pico_add_extra_outputs(macropad15)
